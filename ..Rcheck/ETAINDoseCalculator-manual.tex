\HeaderA{get\_mobilecall\_bt\_headp\_dose}{Calculate RF-EMF dose from mobile calling using Bluetooth headphones (headphone contribution)}{get.Rul.mobilecall.Rul.bt.Rul.headp.Rul.dose}
%
\begin{Description}
Calculates the RF-EMF dose from mobile calling using Bluetooth headphones (headphone contribution).
\end{Description}
%
\begin{Usage}
\begin{verbatim}
get_mobilecall_bt_headp_dose(
  duration,
  headp_prop,
  headp_ear_num,
  params,
  tissue_params
)
\end{verbatim}
\end{Usage}
%
\begin{Arguments}
\begin{ldescription}
\item[\code{duration}] Duration of mobile phone call in seconds per day

\item[\code{headp\_prop}] Proportion of time Bluetooth headphones are used during mobile call

\item[\code{headp\_ear\_num}] If particpant uses 1 or 2 Bluetooth headphones during mobile call

\item[\code{params}] Parameter list

\item[\code{tissue\_params}] Tissue parameter list
\end{ldescription}
\end{Arguments}
%
\begin{Details}
The dose is calculated as:
\deqn{Dose = Duration \times Power \times SAR \times headp_prop \times headp_ear_num}{}

Where:
\begin{itemize}

\item{} \eqn{Duration}{} is the duration of mobile phone call in seconds per day
\item{} \eqn{SAR}{} is the specific absorption rate from Bluetooth in W/kg/W
\item{} \eqn{Power}{} is the Bluetooth output power of the mobile phone during mobile calling mW
\item{} \eqn{headp_prop}{} is the proportion of time Bluetooth headphones are used during mobile calls
\item{} \eqn{headp_ear_num}{} is the number of Bluetooth headphones (1 or 2) used during mobile calls

\end{itemize}

\end{Details}
%
\begin{Value}
Dose from mobile calling using Bluetooth headphones in mJ/kg/day (headphone contribution)
\end{Value}
%

get_mobilecall_dose(

)

\item[\code{headp\_prop}] Proportion of time Bluetooth headphones are used during mobile call while phone is NOT held against ear

\item[\code{urbanicity}] Urbanicity of home / workplace

\item[\code{use\_5g}] TRUE if participant uses 5G services on mobile phone, FALSE if not

\deqn{Dose_{mobilecall} = Dose_{phone} + Dose_{phoneBluetooth} + Dose_{headphonesBluetooth}}{}

\item{} \eqn{Dose_{phoneBluetooth}}{} is the dose from Bluetooth by the phone
\item{} \eqn{Dose_{headphonesBluetooth}}{} is the dose from Bluetooth by the Bluetooth headphones

\end{itemize}

\end{Details}

\begin{Value}

\begin{itemize}

\item{} "brain\_call\_dose" (brain RF-EMF dose from mobile calls in mJ/kg/day)

\end{itemize}

\HeaderA{get\_mobiledata\_pwr}{Calculate mobile data aggregated output power}{get.Rul.mobiledata.Rul.pwr}
%
\begin{Description}
Calculates the output power of the mobile phone during mobile data use in mW
\end{Description}
%
\begin{Usage}
\begin{verbatim}
get_mobiledata_pwr(
  wifi_prop,
  use_5g,
  act_pwr_props,
  travel_time,
  urbanicity,
  params
)
\end{verbatim}
\end{Usage}
%
\begin{Arguments}
\begin{ldescription}
\item[\code{wifi\_prop}] Proportion of time WiFi connection is used for data transfer (vs mobile data)

\item[\code{use\_5g}] TRUE if participant uses 5G services on mobile phone, FALSE if not

\item[\code{act\_pwr\_props}] List with proportion of time spent in low vs low-mid vs mid-high vs high output power activities

\item[\code{travel\_time}] Time spent commuting in seconds per day

\item[\code{urbanicity}] Urbanicity of home / workplace

\item[\code{params}] parameter list
\end{ldescription}
\end{Arguments}
%
\begin{Details}
The mobile phone output power during mobile data use is calculated as:
\deqn{Power = Prop_{WiFi} \times Power_{WiFi} + Prop_{Data} \times Power_{Data}}{}

Where:
\begin{itemize}

\item{} \eqn{Prop_{WiFi}}{}, \eqn{Prop_{Data}}{} are the proportion of time mobile calls are done using WiFi and mobile data, respectively.
\item{} \eqn{Power_{WiFi}}{}, \eqn{Prop_{Data}}{} are the output power (mW) of the mobile phone using WiFi and mobile data respectively.

\end{itemize}

\end{Details}
%
\begin{Value}
Mobile phone output power during mobile data use in mW
\end{Value}
%

\code{\LinkA{get\_mpd\_wifi\_pwr()}{get.Rul.mpd.Rul.wifi.Rul.pwr}}, \code{\LinkA{get\_mpd\_data\_pwr()}{get.Rul.mpd.Rul.data.Rul.pwr}}
\end{SeeAlso}
